-- ============================================================ -- SUPABASE: Hubungkan tracking_token Ambulance -- ============================================================ -- Jalankan di Supabase > SQL Editor. -- Tujuan: -- 1. Satu tracking_token = satu permintaan ambulance. -- 2. Satu tracking_token = satu posisi GPS terakhir ambulance. -- 3. ambulance_locations.tracking_token harus merujuk ke -- ambulance_requests.tracking_token. -- 4. Keduanya aktif untuk Supabase Realtime. -- -- Catatan: bagian FK terakhir akan gagal jika masih ada baris -- ambulance_locations yang token-nya tidak punya pasangan di -- ambulance_requests. Bersihkan data orphan terlebih dahulu. -- ============================================================ -- A. Pastikan token pada tabel permintaan unik. create unique index if not exists ambulance_requests_tracking_token_uidx on public.ambulance_requests (tracking_token); -- B. Pastikan token pada tabel lokasi unik. -- Ini memungkinkan UPSERT GPS berdasarkan tracking_token. create unique index if not exists ambulance_locations_tracking_token_uidx on public.ambulance_locations (tracking_token); -- C. Hubungkan token lokasi ke token permintaan. -- Jika constraint dengan nama ini sudah ada, PostgreSQL akan dilewati. do $$ begin if not exists ( select 1 from pg_constraint c join pg_class t on t.oid = c.conrelid where t.relname = 'ambulance_locations' and c.conname = 'ambulance_locations_tracking_token_fkey' ) then alter table public.ambulance_locations add constraint ambulance_locations_tracking_token_fkey foreign key (tracking_token) references public.ambulance_requests (tracking_token) on update cascade on delete cascade; end if; end $$; -- D. Aktifkan RLS dan izinkan browser memakai tabel. alter table public.ambulance_requests enable row level security; alter table public.ambulance_locations enable row level security; grant select, insert, update on public.ambulance_requests to anon, authenticated; grant select, insert, update on public.ambulance_locations to anon, authenticated; -- E. Policy sederhana untuk aplikasi HTML berbasis anon key. -- Untuk produksi, lebih aman dibatasi dengan Supabase Auth. drop policy if exists "ambulance_requests_select" on public.ambulance_requests; drop policy if exists "ambulance_requests_insert" on public.ambulance_requests; drop policy if exists "ambulance_requests_update" on public.ambulance_requests; drop policy if exists "ambulance_locations_select" on public.ambulance_locations; drop policy if exists "ambulance_locations_insert" on public.ambulance_locations; drop policy if exists "ambulance_locations_update" on public.ambulance_locations; create policy "ambulance_requests_select" on public.ambulance_requests for select to anon, authenticated using (true); create policy "ambulance_requests_insert" on public.ambulance_requests for insert to anon, authenticated with check (true); create policy "ambulance_requests_update" on public.ambulance_requests for update to anon, authenticated using (true) with check (true); create policy "ambulance_locations_select" on public.ambulance_locations for select to anon, authenticated using (true); create policy "ambulance_locations_insert" on public.ambulance_locations for insert to anon, authenticated with check (true); create policy "ambulance_locations_update" on public.ambulance_locations for update to anon, authenticated using (true) with check (true); -- F. Aktifkan Realtime untuk kedua tabel. do $$ begin if not exists ( select 1 from pg_publication_tables where pubname = 'supabase_realtime' and schemaname = 'public' and tablename = 'ambulance_requests' ) then alter publication supabase_realtime add table public.ambulance_requests; end if; if not exists ( select 1 from pg_publication_tables where pubname = 'supabase_realtime' and schemaname = 'public' and tablename = 'ambulance_locations' ) then alter publication supabase_realtime add table public.ambulance_locations; end if; end $$; -- G. Pemeriksaan akhir. select 'ambulance_requests' as table_name, count(*) as rows from public.ambulance_requests union all select 'ambulance_locations', count(*) from public.ambulance_locations; select tablename from pg_publication_tables where pubname = 'supabase_realtime' and tablename in ('ambulance_requests', 'ambulance_locations') order by tablename;
